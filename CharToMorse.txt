#include <xc.inc>

global  UART_Setup, UART_Transmit_Message, UART_Transmit_Morse_Message
global  UART_Transmit_Byte

psect   udata_acs
UART_counter:   ds 1
morse_code:     ds 1
morse_len:      ds 1
temp_char:      ds 1

psect   uart_code,class=CODE

; ============================================================
; UART setup
; ============================================================
UART_Setup:
    bsf     SPEN                        ; serial port enable
    bcf     SYNC                        ; asynchronous
    bcf     BRGH                        ; low speed
    bsf     TXEN                        ; enable transmit
    bcf     BRG16                       ; 8-bit baud generator
    movlw   103                         ; ~9600 baud
    movwf   SPBRG1, A
    bsf     TRISC, PORTC_TX1_POSN, A
    return

; ============================================================
; Send raw bytes
; Entry:
;   FSR2 -> buffer in RAM
;   W    = number of bytes
; ============================================================
UART_Transmit_Message:
    movwf   UART_counter, A
UART_Loop_message:
    movf    POSTINC2, W, A
    call    UART_Transmit_Byte
    decfsz  UART_counter, F, A
    bra     UART_Loop_message
    return

; ============================================================
; Send first N chars from RAM buffer as Morse ASCII via UART
; Entry:
;   FSR2 -> message_buffer in RAM
;   W    = N (0..32)
; ============================================================
UART_Transmit_Morse_Message:
    movwf   UART_counter, A
Morse_Message_Loop:
    movf    POSTINC2, W, A
    call    UART_Transmit_Morse_Char
    decfsz  UART_counter, F, A
    bra     Morse_Message_Loop
    return

; ============================================================
; Input W = valid char: 'A'..'Z', '0'..'9', or ' '
; Output: transmits Morse representation as '.', '-', ' ', '/'
; ============================================================
UART_Transmit_Morse_Char:
    movwf   temp_char, A

    ; space -> "/ "
    movf    temp_char, W, A
    xorlw   ' '
    bz      Morse_Send_Space

    ; digit?
    movf    temp_char, W, A
    sublw   '9'                    ; W = '9' - char
    bc      Maybe_Digit            ; if char <= '9'
    bra     Handle_Letter

Maybe_Digit:
    movf    temp_char, W, A
    sublw   '0'                    ; W = '0' - char
    bnc     Handle_Digit           ; if char >= '0'
    bra     Handle_Letter

Handle_Digit:
    movf    temp_char, W, A
    addlw   -'0'                   ; W = 0..9
    call    Morse_Digit_Table
    bra     Send_Morse_From_Code

Handle_Letter:
    movf    temp_char, W, A
    addlw   -'A'                   ; W = 0..25
    call    Morse_Letter_Table

Send_Morse_From_Code:
    movwf   morse_code, A

    movf    morse_code, W, A
    andlw   0x07                   ; low 3 bits = length
    movwf   morse_len, A

Morse_Symbol_Loop:
    btfsc   morse_code, 7, A
    bra     Send_Dash

Send_Dot:
    movlw   '.'
    call    UART_Transmit_Byte
    bra     Shift_Next_Symbol

Send_Dash:
    movlw   '-'
    call    UART_Transmit_Byte

Shift_Next_Symbol:
    rlcf    morse_code, F, A       ; next symbol into bit7
    decfsz  morse_len, F, A
    bra     Morse_Symbol_Loop

    ; separator between encoded characters
    movlw   ' '
    call    UART_Transmit_Byte
    return

Morse_Send_Space:
    movlw   '/'
    call    UART_Transmit_Byte
    movlw   ' '
    call    UART_Transmit_Byte
    return

; ============================================================
; Send byte in W over UART
; ============================================================
UART_Transmit_Byte:
    btfss   TX1IF
    bra     UART_Transmit_Byte
    movwf   TXREG1, A
    return

; ============================================================
; Morse table for A..Z in flash
; Encoding:
;   bits 7:3 = pattern, left aligned
;   bits 2:0 = length
; dot=0, dash=1
; ============================================================
Morse_Letter_Table:
    addwf   PCL, F, A
    retlw   0x42    ; A  .-
    retlw   0x84    ; B  -...
    retlw   0xA4    ; C  -.-.
    retlw   0x83    ; D  -..
    retlw   0x01    ; E  .
    retlw   0x24    ; F  ..-.
    retlw   0xC3    ; G  --.
    retlw   0x04    ; H  ....
    retlw   0x02    ; I  ..
    retlw   0x74    ; J  .---
    retlw   0xA3    ; K  -.-
    retlw   0x44    ; L  .-..
    retlw   0xC2    ; M  --
    retlw   0x82    ; N  -.
    retlw   0xE3    ; O  ---
    retlw   0x64    ; P  .--.
    retlw   0xD4    ; Q  --.-
    retlw   0x43    ; R  .-.
    retlw   0x03    ; S  ...
    retlw   0x81    ; T  -
    retlw   0x23    ; U  ..-
    retlw   0x14    ; V  ...-
    retlw   0x63    ; W  .--
    retlw   0x94    ; X  -..-
    retlw   0xB4    ; Y  -.--
    retlw   0xC4    ; Z  --..

; ============================================================
; Morse table for 0..9 in flash
; ============================================================
Morse_Digit_Table:
    addwf   PCL, F, A
    retlw   0xFD    ; 0  -----
    retlw   0x7D    ; 1  .----
    retlw   0x3D    ; 2  ..---
    retlw   0x1D    ; 3  ...--
    retlw   0x0D    ; 4  ....-
    retlw   0x05    ; 5  .....
    retlw   0x85    ; 6  -....
    retlw   0xC5    ; 7  --...
    retlw   0xE5    ; 8  ---..
    retlw   0xF5    ; 9  ----.
